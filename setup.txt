-- Function to get the last folder name and truncate it if it exceeds maxLength\r\
local function getFinalFolderName(path, maxLength)\r\
    -- If the path is empty, return root symbol or empty string\r\
    if not path or path == \"\" then\r\
        return \"\"\r\
    end\r\
    -- Extract the substring after the last slash '/'\r\
    local finalName = path:match(\"([^/]+)$\") or path\r\
    -- Check if the length exceeds the limit\r\
    if #finalName > maxLength then\r\
        -- Calculate how many characters from the end we can keep\r\
        -- Subtracting 2 to make room for the \"..\" dots\r\
        local keepLength = maxLength - 2\r\
        -- Prevent negative lengths or errors on very small limits\r\
        if keepLength > 0 then\r\
            finalName = \"..\" .. finalName:sub(#finalName - keepLength + 1)\r\
        else\r\
            finalName = finalName:sub(1, maxLength)\r\
        end\r\
    end\r\
    return finalName\r\
end\r\
